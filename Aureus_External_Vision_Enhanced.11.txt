    };

  function parseJsonLoose(t) {
    if (typeof t !== 'string') return null;
    const s = t.trim().replace(/^```(?:json)?\s*/i, '').replace(/```$/, '').trim();
    try { return JSON.parse(s); } catch (e) { /* fall through */ }
    const a = s.indexOf('{'), b = s.lastIndexOf('}');
    if (a >= 0 && b > a) { try { return JSON.parse(s.slice(a, b + 1)); } catch (e) { /* ignore */ } }
    return null;
  }

  // Accepts the v2 contract (selectedChoice) and the v1 names (suggestedChoice).
  function validateAnalysis(o, req) {
    const bad = (m) => ({ ok: false, error: m });
    if (!o || typeof o !== 'object' || Array.isArray(o)) return bad('not a JSON object');
    const conf = typeof o.confidence === 'string' ? o.confidence.toLowerCase() : '';
    if (!['low', 'medium', 'high'].includes(conf)) return bad('confidence must be low, medium or high');
    if (typeof o.uncertain !== 'boolean') return bad('uncertain must be a boolean');
    const explanation = typeof o.explanation === 'string' ? normText(o.explanation) : '';
    if (!explanation) return bad('explanation is missing');
    if (o.evidence !== undefined && !Array.isArray(o.evidence)) return bad('evidence must be an array when present');
    const evidence = (o.evidence || []).filter((e) => typeof e === 'string').map((e) => normText(e).slice(0, 400)).filter(Boolean).slice(0, 6);
    const rawLetter = o.selectedChoice !== undefined ? o.selectedChoice : o.suggestedChoice;
    const rawText = o.selectedChoiceText !== undefined ? o.selectedChoiceText : o.suggestedChoiceText;
    const letter = typeof rawLetter === 'string' ? rawLetter.trim().toUpperCase() : '';
    let idx = -1;
    if (letter) {
      idx = req.choices.findIndex((c) => c.label === letter);
      if (idx < 0 && !o.uncertain) return bad('selectedChoice is not one of the available choices');
    }
    if (!o.uncertain && idx < 0) return bad('no choice was selected');
    const choice = idx >= 0 ? req.choices[idx] : null;
    if (choice && typeof rawText === 'string' && rawText.trim() && !/^\[image/i.test(choice.text) && choiceNorm(rawText).toLowerCase() !== choice.text.toLowerCase()) {
      return bad('selected text does not match the selected letter');
    }
    return {
      ok: true,
      value: {
        suggestedChoice: choice ? choice.label : '', suggestedChoiceText: choice ? choice.text : '',
        explanation: explanation.slice(0, 1500), confidence: o.uncertain ? 'low' : conf, evidence, uncertain: o.uncertain
      }
    };
  }

  function redact(s, extra) {
    let out = String(s);
    for (const x of extra || []) if (typeof x === 'string' && x.length >= 8) out = out.split(x).join('[redacted]');
    return out
      .replace(/AIza[0-9A-Za-z_-]{20,}/g, '[redacted]')
      .replace(/\bAQ\.[0-9A-Za-z_-]{16,}/g, '[redacted]')
      .replace(/\bgsk_[0-9A-Za-z_-]{16,}/g, '[redacted]')
      .replace(/\bsk-[0-9A-Za-z_-]{16,}/g, '[redacted]')
      .replace(/\bnvapi-[0-9A-Za-z_-]{16,}/gi, '[redacted]')
      .replace(/(Bearer\s+)[^\s"']+/gi, '$1[redacted]')
      .replace(/((?:api[_-]?key|token|authorization)["'\s:=]+)[^\s"',}]{8,}/gi, '$1[redacted]');
  }

  /* ---- model registry helpers ---- */
  function modelCandidates(pid, raw) {
    let ids = [];
    if (pid === 'gemini') {
      ids = ((raw && raw.models) || [])
        .filter((m) => m && typeof m.name === 'string' && Array.isArray(m.supportedGenerationMethods) &&
          (m.supportedGenerationMethods.includes('generateContent') || m.supportedGenerationMethods.includes('streamGenerateContent')))
        .map((m) => m.name.replace(/^models\//, ''))
        .filter((id) => (/^gemini-(\d+(\.\d+)?-)?(flash|pro)(?:-(?:lite|preview|latest|exp|001|\d{2}-\d{2}))?$/i.test(id) || /^gemini-4-argon$/i.test(id)) &&
          !/image|tts|audio|live|transcribe|embedding|computer-use|robotics/i.test(id));
    } else if (pid === 'groq') {
      ids = ((raw && raw.data) || [])
        .filter((m) => m && typeof m.id === 'string' && m.active !== false)
        .map((m) => m.id)
        .filter((id) => validModel(id) && !/whisper|tts|guard|safeguard|orpheus|playai|embed|compound|allam|distil/i.test(id));
    } else if (pid === 'openrouter') {
      // OpenRouter: GET /api/v1/models returns { data: [{ id, architecture, pricing }] }.
      // Keep text-output chat models; drop embeddings, audio, moderation and rerankers.
      ids = ((raw && raw.data) || [])
        .filter((m) => m && typeof m.id === 'string')
        .filter((m) => {
          const out = m.architecture && Array.isArray(m.architecture.output_modalities) ? m.architecture.output_modalities : null;
          return !out || out.includes('text');
        })
        .map((m) => m.id)
        .filter((id) => validModel(id) && !/embed|moderation|whisper|tts|audio|rerank|guard|safety|ocr/i.test(id));
    } else if (pid === 'nvidia') {
      // The NVIDIA Build endpoint exposes its serving ID via /v1/models.
      ids = ((raw && raw.data) || [])
        .filter((m) => m && typeof m.id === 'string')
        .map((m) => m.id)
        .filter((id) => /^nvidia\/nemotron-3\.5-super-vl-120b-a12b$/i.test(id));
    }
    return Array.from(new Set(ids)).slice(0, 100);
  }

  function updateStat(stats, id, ms, ok) {
    const s = stats[id] || { ema: 0, n: 0, fails: 0 };
    if (ok) { s.ema = s.n ? s.ema * 0.7 + ms * 0.3 : ms; s.n += 1; s.fails = 0; } else { s.fails += 1; }
    stats[id] = s;
    return s;
  }

  // Order: measured+reliable (lowest latency first), then unmeasured by prior, then models with repeated failures.
  function rankModels(pid, ids, stats, unavailable, now) {
    const st = stats || {}, un = unavailable || {};
    const usable = ids.filter((id) => !(un[id] && now - un[id] < 3600000));
    const bad = (id) => st[id] && st[id].fails >= 3;
    const measured = (id) => st[id] && st[id].n > 0 && !bad(id);
    // Rough speed prior, used only until real timings arrive.
    const gemVer = (id) => { const m = /^gemini-(\d+)(?:\.(\d+))?/.exec(id); return m ? Number(m[1]) * 1000 + Number(m[2] || 0) * 10 : -1; };
    const maxVer = pid === 'gemini' ? Math.max.apply(null, ids.map(gemVer).concat([0])) : 0;
    const prior = (id) => {
      if (pid === 'gemini') {
        const v = gemVer(id);
        const base = v < 0 ? maxVer : v;              // "gemini-flash-latest" follows the newest version
        const lite = /-lite$/.test(id) ? 0 : 5;         // full flash preferred over lite
        const pro = /-pro(-|$)/.test(id) ? -25 : 0;     // pro is slower to answer
        const prev = /-preview$/.test(id) ? -1 : 0;
        return base + lite + pro + prev;
      }
      if (pid === 'openrouter') {
        // Ordering only: OpenRouter's own router first, then small/fast general models.
        // This is not a capability claim; the live /models list decides availability.
        const s = id.toLowerCase();
        if (s === 'openrouter/auto') return 60;
        let score = 0;
        if (/flash|mini|haiku|8b|7b|small|instant|turbo/.test(s)) score += 20;
        if (/:free$/.test(s)) score += 6;
        if (/reasoning|thinking|r1|qwq|o1|o3|o4/.test(s)) score -= 15;
        if (/pro|large|70b|405b|235b/.test(s)) score -= 6;
        return score;
      }
      return (DOC_SPEED_TPS[pid] && DOC_SPEED_TPS[pid][id]) || 0;
    };
    return usable.slice().sort((a, b) => {
      if (!!bad(a) !== !!bad(b)) return bad(a) ? 1 : -1;
      if (!!measured(a) !== !!measured(b)) return measured(a) ? -1 : 1;
      if (measured(a)) return st[a].ema - st[b].ema;
      return prior(b) - prior(a);
    });
  }

  const isAcceptable = (rec, policy) => rec.status === 'ok' && !!rec.value && !rec.value.uncertain &&
    CONF_RANK[rec.value.confidence] >= CONF_RANK[policy.minConfidence];

  // recs: finished provider records; expected: number of providers launched.
  function decideOutcome(recs, expected, policy) {
    const fin = recs.slice();
    const ok = fin.filter((r) => r.status === 'ok' && r.value);
    const good = ok.filter((r) => isAcceptable(r, policy)).sort((a, b) => a.latencyMs - b.latencyMs);
    const all = fin.length >= expected;
    const out = (o) => Object.assign({ done: true, finished: fin }, o);
    if (policy.mode === 'verify') {
      // Wait for every participant, then group normalized answer letters.
      if (!all) return { done: false };
      const threshold = Math.max(2, Math.min(3, policy.threshold || 2));
      const votes = new Map();
      for (const r of good) {
        const ch = String(r.value.suggestedChoice || '').toUpperCase();
        if (!ch) continue;
        const g = votes.get(ch) || [];
        g.push(r); votes.set(ch, g);
      }
      const groups = Array.from(votes.values()).sort((a, b) => b.length - a.length || a[0].latencyMs - b[0].latencyMs);
      if (!groups.length) {
        if (ok.length) {
          ok.sort((a, b) => (CONF_RANK[b.value.confidence] - CONF_RANK[a.value.confidence]) || (a.latencyMs - b.latencyMs));
          return out({ kind: 'inconclusive', best: ok[0] });
        }
        return out({ kind: 'failed' });
      }
      const top = groups[0], topLetter = String(top[0].value.suggestedChoice).toUpperCase();
      const distinct = groups.filter((g) => String(g[0].value.suggestedChoice).toUpperCase() !== topLetter);
      if (top.length >= threshold) {
        return out({ kind: 'accepted', winner: top[0], verified: true, agreed: top.length > 1, agreement: { letter: topLetter, count: top.length, of: good.length, groups } });
      }
      if (distinct.length) {
        return out({ kind: 'conflict', conflict: good, agreement: { letter: topLetter, count: top.length, of: good.length, groups } });
      }
      return out({ kind: 'unverified', best: top[0], agreement: { letter: topLetter, count: top.length, of: good.length, groups } });
    }
    if (policy.mode === 'agree') {
      if (!all) return { done: false };
      if (good.length >= 2) {
        const letters = new Set(good.map((r) => r.value.suggestedChoice));
        return letters.size === 1 ? out({ kind: 'accepted', winner: good[0], agreed: true }) : out({ kind: 'conflict', conflict: good });
      }
      if (good.length === 1) return out({ kind: 'accepted', winner: good[0], single: expected > 1 });
    } else if (good.length) {
      return out({ kind: 'accepted', winner: good[0] });
    }
    if (!all) return { done: false };
    if (ok.length) {
      ok.sort((a, b) => (CONF_RANK[b.value.confidence] - CONF_RANK[a.value.confidence]) || (a.latencyMs - b.latencyMs));
      return out({ kind: 'inconclusive', best: ok[0] });
    }
    return out({ kind: 'failed' });
  }

  // Runs every provider concurrently; resolves as soon as decideOutcome is satisfied.
  // runner(pid, controller) must resolve to a record and should honour controller.cancelled/abort().
  function raceProviders(pids, runner, policy, parentCtl) {
    const subs = pids.map((pid) => ({ pid, c: { cancelled: false, abort() {} } }));
    const roles = {};
    parentCtl.abort = () => subs.forEach((s) => { s.c.cancelled = true; s.c.abort(); });
    return new Promise((resolve) => {
      let settled = false;
      const finished = [];
      const conclude = (d) => {
        settled = true;
        for (const s of subs) {
          const rec = finished.find((r) => r.provider === s.pid);
          if (d.winner && d.winner.provider === s.pid) roles[s.pid] = 'winner';
          else if (rec) roles[s.pid] = 'completed, not selected';
          else { roles[s.pid] = 'cancel requested (client-side abort)'; s.c.cancelled = true; s.c.abort(); }
        }
        resolve(Object.assign({}, d, { roles }));
      };
      if (!subs.length) { conclude({ done: true, kind: 'failed', finished: [] }); return; }
      subs.forEach((s) => {
        Promise.resolve().then(() => runner(s.pid, s.c))
          .catch((e) => ({ provider: s.pid, status: 'error', failure: 'internal', message: String((e && e.message) || e), latencyMs: 0 }))
          .then((rec) => {
            if (settled) return;
            if (parentCtl.cancelled) { settled = true; resolve({ done: true, kind: 'cancelled', finished, roles }); return; }
            finished.push(rec);
            const d = decideOutcome(finished, subs.length, policy);
            if (d.done) conclude(d);
          });
      });
    });
  }

    edit: { question: '', choices: '', notes: '', images: [], dirty: false },
    analysis: { status: 'idle', key: '', sig: '', result: null, error: null, info: null, fromCache: false, highlight: '', attempt: 0 },
    auto: { status: 'idle', message: 'Auto-select is off.', at: 0, sig: '' },
    run: null, test: {},
    history: validateHistory(readJson(K.history)),
    tab: 'overview'
  };
  let current = null; // in-flight analysis controller
  let lastMap = null, autoTimer = 0, autoAttempt = 0;

  /* ====================================================================
   * Question detection
   * Hooks verified against Wayground's app bundle (new gameplay MCQ):
   *   [data-testid="question-query-text"], [data-testid="question-option"],
   *   [data-testid="option-text"], [data-testid="mcq"], .option-container,
   *   .query-container, .question-section.
   * Older/generic selectors remain as fallbacks.
   * ==================================================================== */
  const diag = (msg) => { if (settings.diagnostics) log('diag', msg); };
  const qsaSafe = (sel) => { try { return Array.from(document.querySelectorAll(sel)); } catch (e) { return []; } };
  const inOurUi = (el) => !!(el && el.closest && el.closest('#' + HOST_ID));
  const isVisible = (el) => {
    if (!el || !el.isConnected) return false;
    const r = el.getBoundingClientRect();
    if (r.width < 4 || r.height < 4) return false;
    const cs = getComputedStyle(el);
    return cs.display !== 'none' && cs.visibility !== 'hidden' && parseFloat(cs.opacity) > 0.05;
  };
  const elText = (el) => normText(el.innerText || el.textContent || '');

  // Answer tiles on Wayground render an option number before the text; reading the text node
  // that owns data-testid="option-text" skips that badge, and the fallback strips it manually.
  function choiceText(el) {
    const inner = el.querySelector('[data-testid="option-text"],[data-testid="image-caption"]');
    if (inner) {
      const s = normText(inner.innerText || inner.textContent || '');
      if (s) return s;
    }
    const kh = el.querySelector('[data-functional-selector^="question-choice-text-"]');
    if (kh) {
      const s = normText(kh.innerText || kh.textContent || '');
      if (s) return s;
    }
    const s = elText(el);
    try {
      if (el.matches(WG_OPTION)) {
        const stripped = s.replace(/^\d{1,2}(?:\s+|$)/, '').trim();
        if (stripped) return stripped;
      }
    } catch (e) { /* ignore */ }
    if (s && !/^\d{1,2}$/.test(s)) return s;
    const alt = el.querySelector('img[alt]');
    if (alt && normText(alt.getAttribute('alt') || '').length) return normText(alt.getAttribute('alt')).slice(0, MAX_C);
    return '';
  }

  // Keep card-level elements: drop wrappers holding 2+ candidates, then drop candidates nested in another.
  function cardLevel(c) {
    const noWrap = c.filter((el) => c.reduce((n, o) => n + (o !== el && el.contains(o) ? 1 : 0), 0) < 2);
    return noWrap.filter((el) => !noWrap.some((o) => o !== el && o.contains(el)));
  }

  // Wayground marks the gameplay answer tiles with data-testid="question-option" and gives
  // the text its own data-testid="option-text" (image answers use "image-caption").
  // Kahoot uses data-functional-selector="answer-N" and "question-choice-text-N".
  const WG_OPTION = '[data-testid="question-option"],[data-testid="classification-option"],.option-container';
  const CHOICE_STRATEGIES = [
    ['wayground-option', () => qsaSafe('[data-testid="question-option"],[data-testid="classification-option"]')],
    ['kahoot-answer', () => qsaSafe('[data-functional-selector^="answer-"]')
      .filter((el) => /^answer-\d+$/.test(el.getAttribute('data-functional-selector') || ''))],
    ['aria-roles', () => qsaSafe('[role="radio"],[role="checkbox"],[role="option"]')],
    ['test-id-option', () => qsaSafe('[data-testid*="option" i],[data-testid*="choice" i],[data-cy*="option" i],[data-cy*="choice" i]')],
    ['class-option', () => qsaSafe('[class*="option" i],[class*="choice" i]')],
    ['test-id-answer', () => qsaSafe('[data-testid*="answer" i],[data-cy*="answer" i],[data-test*="answer" i]')],
    ['class-answer', () => qsaSafe('[class*="answer" i],[class*="responses" i]')]
  ];

  const MAX_VISION_IMAGES = 5;
  const MAX_CAPTURE_IMAGE_REFS = 10;
  const MAX_VISION_DOWNLOAD_BYTES = 8 * 1024 * 1024;
  const MAX_VISION_DIMENSION = 1536;
  const VISION_DOWNLOAD_TIMEOUT_MS = 12000;

  function hasMeaningfulImage(el) {
    if (!el || !el.isConnected) return false;
    // Includes CSS background-image options, which some Wayground question types use.
    return visualNodesIn(el).some((node) => {
      if (inOurUi(node)) return false;
      const r = node.getBoundingClientRect();
      const w = Math.max(r.width || 0, node.naturalWidth || node.width || 0);
      const h = Math.max(r.height || 0, node.naturalHeight || node.height || 0);
      return w >= 32 && h >= 24 && validVisionSource(imageSourceFromNode(node));
    });
  }

  function isSafeRemoteImageUrl(src) {
    try {
      const url = new URL(src);
      if (url.protocol !== 'https:' || url.username || url.password) return false;
      const host = url.hostname.toLowerCase().replace(/^\[|\]$/g, '');
      if (!host || !host.includes('.') || host === 'localhost' || /\.(?:localhost|local|lan|home|internal|test|invalid)$/.test(host)) return false;
      // Refuse literal IP addresses (including private/reserved IPv4 and IPv6)
      // to keep this privileged cross-origin downloader away from local services.
      if (host.includes(':') || /^(?:\d{1,3}\.){3}\d{1,3}$/.test(host)) return false;
      return true;
    } catch (e) { return false; }
  }

  function validVisionSource(src) {
    if (typeof src !== 'string') return false;
    const value = src.trim();
    if (/^data:image\//i.test(value)) return value.length <= Math.ceil(MAX_VISION_DOWNLOAD_BYTES * 1.5);
    if (/^blob:/i.test(value)) return value.length <= 12000;
    return value.length <= 12000 && isSafeRemoteImageUrl(value);
  }

  function imageSourceFromNode(node) {
    if (!node) return '';
    if (node.tagName === 'IMG') return cleanImageUrl(node.currentSrc || node.src || node.getAttribute('src') || '');
    if (node.tagName === 'CANVAS') {
      try { return node.toDataURL('image/png'); } catch (e) { return ''; }
    }
    if (node.tagName && node.tagName.toLowerCase() === 'svg') {
      try {
        const markup = new XMLSerializer().serializeToString(node);
        if (markup.length < 2_000_000) return 'data:image/svg+xml;charset=utf-8,' + encodeURIComponent(markup);
      } catch (e) { /* ignore SVGs that cannot be serialized */ }
    }
    try {
      const bg = getComputedStyle(node).backgroundImage || '';
      const m = /url\([\"']?(.*?)[\"']?\)/i.exec(bg);
      if (m && m[1]) return cleanImageUrl(m[1]);
    } catch (e) { /* ignore */ }
    return '';
  }

  function visualNodesIn(root) {
    if (!root) return [];
    const all = [];
    try {
      if (root.matches && root.matches('img,canvas,svg,[style*="background-image"]')) all.push(root);
      else {
        // The image may be applied to the card with a CSS class rather than an inline style.
        try { if ((getComputedStyle(root).backgroundImage || '').includes('url(')) all.push(root); } catch (e) { /* ignore */ }
      }
      all.push(...Array.from(root.querySelectorAll('img,canvas,svg,[style*="background-image"]')));
      // Some answer cards use CSS classes instead of inline background-image styles.
      for (const el of Array.from(root.querySelectorAll('*'))) {
        if (all.includes(el)) continue;
        try { if ((getComputedStyle(el).backgroundImage || '').includes('url(')) all.push(el); } catch (e) { /* ignore */ }
      }
    } catch (e) { return []; }
    return all.filter((node) => {
      if (!isVisible(node) || inOurUi(node)) return false;
      const r = node.getBoundingClientRect();
      return r.width >= 24 && r.height >= 20;
    });
  }

  function collectQuestionImageRefs(questionEl, choiceEls) {
    const out = [], seen = new Set();
    const addFromRoot = (root, labelPrefix, maxForRoot) => {
      for (const node of visualNodesIn(root)) {
        if (choiceEls.some((ch) => ch === node || ch.contains(node))) continue;
        const src = imageSourceFromNode(node);
        if (!validVisionSource(src) || seen.has(src)) continue;
        // Skip tiny UI glyphs and decorative pictures; retain large diagram/SVG content.
        const r = node.getBoundingClientRect();
        const w = Math.max(r.width || 0, node.naturalWidth || node.width || 0);
        const h = Math.max(r.height || 0, node.naturalHeight || node.height || 0);
        if (w < 32 || h < 24) continue;
        seen.add(src);
        out.push({ label: labelPrefix + (out.length + 1), src });
        if (out.length >= maxForRoot || out.length >= MAX_CAPTURE_IMAGE_REFS) return;
      }
    };

    // Climb only as far as needed to include an image paired with the question text.
    let scope = questionEl;
    for (let depth = 0; scope && depth < 4 && out.length < MAX_CAPTURE_IMAGE_REFS; depth++) {
      const before = out.length;
      addFromRoot(scope, 'Question image ', 2);
      if (out.length > before) break;
      const parent = scope.parentElement;
      if (!parent) break;
      // The parent may hold both prompt and options; option descendants are excluded above.
      scope = parent;
    }

    // Label images by answer letter so visual multiple-choice options remain interpretable.
    choiceEls.forEach((choiceEl, index) => {
      if (out.length >= MAX_CAPTURE_IMAGE_REFS) return;
      const letter = String.fromCharCode(65 + index);
      const nodes = visualNodesIn(choiceEl);
      for (const node of nodes) {
        const src = imageSourceFromNode(node);
        if (!validVisionSource(src) || seen.has(src)) continue;
        const r = node.getBoundingClientRect();
        const w = Math.max(r.width || 0, node.naturalWidth || node.width || 0);
        const h = Math.max(r.height || 0, node.naturalHeight || node.height || 0);
        if (w < 32 || h < 24) continue;
        seen.add(src);
        out.push({ label: 'Answer ' + letter + ' image', src });
        if (out.length >= MAX_CAPTURE_IMAGE_REFS) break;
      }
    });
    return out.slice(0, MAX_CAPTURE_IMAGE_REFS);
  }

  function findChoices() {
    const strategies = settings.customChoiceSelector
      ? [['custom-selector', () => qsaSafe(settings.customChoiceSelector)]]
      : CHOICE_STRATEGIES;
    for (const [name, fn] of strategies) {
      const raw = fn().filter((el) => !inOurUi(el) && isVisible(el)).slice(0, 200);
      const els = cardLevel(raw).filter((el) => {
        const t = choiceText(el);
        return (t.length > 0 && t.length <= MAX_C) || hasMeaningfulImage(el);
      });
      diag('choices via ' + name + ': raw=' + raw.length + ' card-level=' + els.length);
      if (els.length >= 2 && els.length <= 8) return { name, els };
    }
    return null;
  }

  const QUESTION_SELECTORS = '[data-testid="question-query-text"],[data-testid="advanced-rte-query"],[data-functional-selector="block-title"],[data-testid*="question" i],[class*="question" i],[role="heading"],h1,h2,h3';
  const COUNTER = /^(question\s*)?\d+\s*(of|\/)\s*\d+$/i;
  function findQuestion(choiceEls) {
    const first = choiceEls[0];
    let c = settings.customQuestionSelector ? qsaSafe(settings.customQuestionSelector) : qsaSafe(QUESTION_SELECTORS);
    c = c.filter((el) => !inOurUi(el) && isVisible(el) && !choiceEls.some((ch) => ch.contains(el) || el.contains(ch)) &&
      (first.compareDocumentPosition(el) & Node.DOCUMENT_POSITION_PRECEDING));
    c = c.filter((el) => {
      const t = elText(el);
      return (t.length >= 3 && t.length <= MAX_Q && !COUNTER.test(t)) || hasMeaningfulImage(el);
    });
    c = c.filter((el) => !c.some((o) => o !== el && el.contains(o)));
    if (!c.length) {
      // Some image-led questions put the only prompt in an image or CSS-background card above the choices.
      const visuals = visualNodesIn(document.body || document.documentElement).filter((el) => {
        const src = imageSourceFromNode(el);
        return el !== document.body && el !== document.documentElement && validVisionSource(src) &&
          !choiceEls.some((ch) => ch === el || ch.contains(el) || el.contains(ch)) &&
          (first.compareDocumentPosition(el) & Node.DOCUMENT_POSITION_PRECEDING);
      });
      return visuals.length ? visuals[visuals.length - 1] : null;
    }
    const strong = c.filter((el) => {
      const fsel = el.getAttribute && (el.getAttribute('data-functional-selector') || '');
      const cls = el.className && el.className.toString ? el.className.toString() : '';
      return fsel === 'block-title' || /question/i.test((el.getAttribute('data-testid') || '') + ' ' + cls);
    });
    if (strong.length) return strong.reduce((a, b) => (elText(b).length >= elText(a).length ? b : a));
    // No obvious question container: prefer a block that reads like a question.
    const asked = c.filter((el) => /\?/.test(elText(el)));
    if (asked.length) return asked.reduce((a, b) => (elText(b).length >= elText(a).length ? b : a));
    return c[c.length - 1];
  }

  function detectNow() {
    const out = { ok: false, status: 'none', strategy: '', reason: '', question: '', choices: [], qEl: null, cEls: [], imageRefs: [], sig: '' };
    const ch = findChoices();
    if (!ch) { out.reason = 'No answer choices were recognized on this page. Use the manual fields below.'; return out; }
    const q = findQuestion(ch.els);
    if (!q) { out.status = 'unsupported'; out.reason = 'Answer choices were found but no question text or question image was recognized.'; return out; }
    const rawQuestion = elText(q);
    const qt = rawQuestion.length >= 3 ? rawQuestion : (hasMeaningfulImage(q) ? '[Question image: inspect the attached picture]' : rawQuestion);
    const cts = ch.els.map((el, i) => {
      const t = choiceNorm(choiceText(el));
      return t || (hasMeaningfulImage(el) ? '[Image answer ' + String.fromCharCode(65 + i) + ']' : '');
    });
    if (qt.length < 8) { out.reason = 'The question text is too short and no usable question image was found.'; return out; }
    if (cts.some((t) => !t)) { out.reason = 'A choice is empty (content may still be loading).'; return out; }
    if (new Set(cts.map((t) => t.toLowerCase())).size !== cts.length) { out.reason = 'Two visible choices have identical text.'; return out; }
    const imageRefs = collectQuestionImageRefs(q, ch.els);
    return { ok: true, status: 'ok', strategy: ch.name, reason: '', question: qt, choices: cts, qEl: q, cEls: ch.els, imageRefs, sig: sigOf(qt, cts) };
  }

  let pendingSig = null, committedSig = '', observer = null, scanRetryTimer = 0, scanMisses = 0;
  const SCAN_RETRY_MS = [800, 1600, 3000, 5000, 8000];

  // A scan that runs before the page finished rendering (fonts, hydration, lazy images)
  // used to leave the panel stuck on "not detected" forever. Failed scans now retry a
  // few times with backoff, until the page settles or the budget is spent.
  function retryScan() {
    if (scanRetryTimer || scanMisses >= SCAN_RETRY_MS.length) return;
    const wait = SCAN_RETRY_MS[scanMisses++];
    scanRetryTimer = later(() => {
      scanRetryTimer = 0;
      if (settings.enabled && settings.autoDetect) scan();
    }, wait);
  }
  function scan() {
    if (!settings.enabled || !settings.autoDetect) return;
    const d = detectNow();
    const sig = d.ok ? d.sig : 'NONE';
    if (sig !== pendingSig) { pendingSig = sig; later(scan, 350); return; } // wait until content is stable
    if (sig === committedSig) { if (!d.ok) retryScan(); return; }
    committedSig = sig;
    scanMisses = 0;
    commitDetection(d, false);
    if (!d.ok) retryScan();
  }
  let scanTimer = 0;
  const scanSoon = () => { if (scanTimer) clearTimeout(scanTimer); scanTimer = setTimeout(scan, Math.max(100, settings.detectDebounceMs || 400)); };
  const scheduleScan = () => { scanMisses = 0; scanSoon(); };
  function scanNow() {
    const d = detectNow();
    pendingSig = committedSig = d.ok ? d.sig : 'NONE';
    commitDetection(d, true);
  }

  function commitDetection(d, force) {
    const changed = d.sig !== S.detect.sig || d.ok !== S.detect.ok;
    if (!changed && !force) {
      if (d.reason !== S.detect.reason) { S.detect.reason = d.reason; S.detect.status = d.status; ui.refresh('detect'); }
      return;
    }
    if (changed) {
      cancelAnalysis('The question changed.');
      clearHighlights();
      S.analysis = blankAnalysis();
    }
    S.detect = { ok: d.ok, status: d.status, strategy: d.strategy, reason: d.reason, question: d.question, choices: d.choices, sig: d.sig };
    if (d.ok && (force || changed)) { S.edit.question = d.question; S.edit.choices = d.choices.join('\n'); S.edit.dirty = false; }
    else if (!d.ok && !S.edit.dirty) { S.edit.question = ''; S.edit.choices = ''; }
    if (!d.ok) diag('detection: ' + d.status + ' - ' + d.reason);
    ui.refresh('detect');
    if (d.ok && changed && settings.autoAnalyze && settings.acknowledged) runAnalysis(false);
  }

  function startDetection() {
    if (observer || !settings.enabled || !settings.autoDetect) return;
    observer = new MutationObserver((muts) => {
      for (const m of muts) {
        const t = m.target.nodeType === 1 ? m.target : m.target.parentElement;
        if (t && !inOurUi(t)) { scheduleScan(); return; }
      }
    });
    observer.observe(document.body || document.documentElement, { childList: true, subtree: true, characterData: true });
    scheduleScan();
  }
  function stopDetection() {
    if (observer) { observer.disconnect(); observer = null; }
    scanSoon.cancel();
    if (scanRetryTimer) { clearTimeout(scanRetryTimer); scanRetryTimer = 0; }
    scanMisses = 0; pendingSig = null;
  }

  /* ====================================================================
   * Page highlighting (outline classes only; never clicks anything)
   * ==================================================================== */
  let highlighted = [];
  function updatePageStyle() {
    const h = settings.highlight;
    const glow = h.glow ? 'box-shadow:0 0 18px 2px var(--wsa-c)!important;' : '';
    setPageCss(
      '.wsa-hl-ok,.wsa-hl-unsure,.wsa-hl-na{outline:3px solid var(--wsa-c)!important;outline-offset:2px!important;' + glow + '}' +
      '.wsa-hl-ok{--wsa-c:' + h.okColor + '}.wsa-hl-unsure{--wsa-c:' + h.uncertainColor + '}' +
      '.wsa-hl-na{--wsa-c:' + h.unavailableColor + ';outline-style:dashed!important}' +
      '.wsa-rel{position:relative!important}' +
      '.wsa-badge::after{position:absolute;top:-11px;right:10px;padding:1px 8px;border-radius:999px;font:600 11px/18px system-ui,sans-serif;background:var(--wsa-c);color:#06101a;pointer-events:none;z-index:2147482000}' +
      '.wsa-hl-ok.wsa-badge::after{content:"AI suggestion \\00B7 unverified"}' +
      '.wsa-hl-unsure.wsa-badge::after{content:"AI unsure \\00B7 unverified"}' +
      '.wsa-hl-na.wsa-badge::after{content:"AI: no suggestion"}');
  }
  let pageSheet = null;
  function setPageCss(css) {
    try {
      if (!pageSheet) { pageSheet = new CSSStyleSheet(); document.adoptedStyleSheets = document.adoptedStyleSheets.concat([pageSheet]); }
      pageSheet.replaceSync(css);
      return;
    } catch (e) { pageSheet = null; /* fall back to a <style> element */ }
    let st = document.getElementById(PAGE_STYLE_ID);
    if (!st) { st = document.createElement('style'); st.id = PAGE_STYLE_ID; (document.head || document.documentElement).append(st); }
    st.textContent = css;
  }
  function removePageCss() {
    try { if (pageSheet) document.adoptedStyleSheets = document.adoptedStyleSheets.filter((x) => x !== pageSheet); } catch (e) { /* ignore */ }
    pageSheet = null;
    const st = document.getElementById(PAGE_STYLE_ID); if (st) st.remove();
  }
  function clearHighlights() {
    for (const el of highlighted) el.classList.remove('wsa-hl-ok', 'wsa-hl-unsure', 'wsa-hl-na', 'wsa-badge', 'wsa-rel');
    highlighted = [];
  }
  function mark(el, cls) {
    if (!el) return;
    el.classList.add(cls);
    if (settings.highlight.badge) {
      el.classList.add('wsa-badge');
      if (getComputedStyle(el).position === 'static') el.classList.add('wsa-rel');
    }
    highlighted.push(el);
  }
  function applyHighlight() {
    clearHighlights();
    const a = S.analysis;
    a.highlight = '';
    if (!settings.highlight.enabled) { a.highlight = 'Highlighting is turned off.'; return; }
    if (a.status !== 'done' && a.status !== 'error') return;
    const d = detectNow();
    if (!d.ok) { a.highlight = 'Not highlighted: the page choices could not be re-detected.'; return; }
    if (d.sig !== a.sig) { a.highlight = 'Not highlighted: the visible question or choices differ from the text that was analyzed.'; return; }
      const r = a.result;
      const idx0 = r && r.suggestedChoice ? r.suggestedChoice.charCodeAt(0) - 65 : -1;
      lastMap = r && idx0 >= 0 && d.cEls[idx0] ? { sig: a.sig, idx: idx0, text: r.suggestedChoiceText, choice: r.suggestedChoice } : null;
      if (a.status === 'done' && r) {
      const idx = r.suggestedChoice ? r.suggestedChoice.charCodeAt(0) - 65 : -1;
      const el = idx >= 0 ? d.cEls[idx] : null;
      if (el && choiceNorm(choiceText(el)).toLowerCase() === r.suggestedChoiceText.toLowerCase()) {
        mark(el, r.uncertain ? 'wsa-hl-unsure' : 'wsa-hl-ok');
        a.highlight = r.uncertain ? 'Tentative choice outlined as uncertain.' : 'Suggested choice outlined. Nothing was clicked.';
        return;
      }
    }
    mark(d.qEl, 'wsa-hl-na');
    a.highlight = 'No choice outlined (no reliable suggestion).';
  }

  /* ---- optional answer selection (authorized testing only) ----
   * Disabled by default and requires an explicit acknowledgement. Mapping is
   * rebuilt from the live page every attempt (never from the highlight state),
   * and it only acts when the question signature and the option text still
   * match. It never submits a quiz and never guesses an ambiguous mapping. */
  function clearAutoTimer() { if (autoTimer) { clearTimeout(autoTimer); autoTimer = 0; } }
  function autoSay(status, message) {
    S.auto = { status, message, at: Date.now(), sig: S.analysis.sig || '' };
    try { ui.refresh('analysis'); } catch (e) { /* UI not mounted yet */ }
  }
  function mapSuggestion(d, r) {
    if (!d || !d.ok || !r || !r.suggestedChoiceText) return { ok: false, message: 'no reliable suggestion text to map' };
    const target = choiceNorm(r.suggestedChoiceText).toLowerCase();
    if (!target) return { ok: false, message: 'the suggested answer text is empty' };
    const texts = d.choices.map((t) => choiceNorm(t).toLowerCase());
    const hits = [];
    texts.forEach((t, i) => { if (t === target) hits.push(i); });
    if (hits.length === 1) return { ok: true, idx: hits[0], el: d.cEls[hits[0]] };
    if (hits.length > 1) return { ok: false, message: 'two visible choices share the suggested text' };
    // Tolerate punctuation/spacing differences ("2,400" vs "2400") only when
    // exactly one choice matches, so an ambiguous mapping is still refused.
    const key = (s) => s.replace(/[^a-z0-9]/g, '');
    const kt = key(target);
    if (kt.length >= 2) {
      const loose = [];
      texts.forEach((t, i) => { if (key(t) === kt) loose.push(i); });
      if (loose.length === 1) return { ok: true, idx: loose[0], el: d.cEls[loose[0]] };
    }
    return { ok: false, message: 'the visible options no longer match the analyzed text (page changed)' };
  }
  function clickChoice(el) {
    const target = (el.closest && el.closest('button,[role="button"],[role="radio"],[role="option"],a')) || el;
    try { target.scrollIntoView({ block: 'center', inline: 'nearest' }); } catch (e) { /* ignore */ }
    try { target.focus({ preventScroll: true }); } catch (e) { /* ignore */ }
    const rect = target.getBoundingClientRect();
    const base = {
      bubbles: true, cancelable: true, composed: true, view: window, button: 0,
      clientX: Math.round(rect.left + rect.width / 2), clientY: Math.round(rect.top + rect.height / 2)
    };
    const fire = (type, Ctor) => { try { target.dispatchEvent(new Ctor(type, base)); } catch (e) { /* ignore */ } };
    // Pointer sequence first: some quiz front ends act on pointerdown, not click.
    if (typeof PointerEvent === 'function') {
      ['pointerover', 'pointerenter', 'pointermove', 'pointerdown'].forEach((t) => fire(t, PointerEvent));
    }
    ['mouseover', 'mousemove', 'mousedown'].forEach((t) => fire(t, MouseEvent));
    if (typeof PointerEvent === 'function') fire('pointerup', PointerEvent);
    fire('mouseup', MouseEvent);
    target.click();
    return target;
  }
  function maybeAutoSelect() {
    clearAutoTimer();
    autoAttempt = 0;
    if (!settings.autoSelect.enabled || !settings.autoSelectAck) { autoSay('idle', 'Auto-select is off.'); return; }
    const a = S.analysis, r = a.result;
    if (a.status !== 'done' || !r || !r.suggestedChoice) { autoSay('idle', 'Waiting for a usable suggestion.'); return; }
    const wait = Math.max(500, settings.autoSelect.delayMs);
    autoSay('waiting', 'Will click ' + r.suggestedChoice + '. ' + truncate(r.suggestedChoiceText, 60) + ' in ' + (wait / 1000).toFixed(1) + ' s.');
    autoTimer = setTimeout(() => { autoTimer = 0; attemptAutoSelect(0); }, wait);
  }
  function attemptAutoSelect(attempt) {
    autoAttempt = attempt;
    if (!settings.autoSelect.enabled || !settings.autoSelectAck) { autoSay('idle', 'Auto-select was turned off.'); return false; }
    const a = S.analysis, r = a.result;
    if (!r || !r.suggestedChoice) { autoSay('refused', 'No suggestion to click.'); return false; }
    const d = detectNow();
    if (!d.ok) {
      if (attempt < 2) { clearAutoTimer(); autoTimer = setTimeout(() => { autoTimer = 0; attemptAutoSelect(attempt + 1); }, 450); return false; }
      autoSay('refused', 'Could not re-read the answer choices.'); log('warn', 'auto-select: choices not detectable'); return false;
    }
    if (d.sig !== a.sig) { autoSay('refused', 'The question or choices changed after analysis.'); log('warn', 'auto-select: question changed'); return false; }
    const map = mapSuggestion(d, r);
    if (!map.ok) {
      if (attempt < 2) { clearAutoTimer(); autoTimer = setTimeout(() => { autoTimer = 0; attemptAutoSelect(attempt + 1); }, 450); return false; }
      autoSay('refused', map.message); log('warn', 'auto-select: ' + map.message); return false;
    }
    if (settings.autoSelect.confirm && !confirm('Select answer ' + r.suggestedChoice + ': "' + r.suggestedChoiceText + '"?')) {
      autoSay('cancelled', 'Cancelled at the confirmation prompt.'); log('info', 'auto-select cancelled at confirmation'); return false;
    }
    const d2 = detectNow(), map2 = mapSuggestion(d2, r);
    if (!d2.ok || d2.sig !== a.sig || !map2.ok || map2.el !== map.el || !map.el.isConnected) {
      autoSay('refused', 'The page changed before the click.'); log('warn', 'auto-select: page changed before acting'); return false;
    }
    try {
      clickChoice(map.el);
      autoSay('clicked', 'Clicked ' + r.suggestedChoice + '. ' + truncate(r.suggestedChoiceText, 60) + '.');
      log('info', 'auto-select clicked ' + r.suggestedChoice + ' (authorized testing)');
      return true;
    } catch (e) {
      autoSay('refused', 'The click was blocked by the page.');
      log('warn', 'auto-select failed: ' + (e && e.message ? e.message : 'click error'));
      return false;
    }
  }

    }

  }

  function modelPlan(pid, needsVision) {
    const cfg = settings.providers[pid], ms = modelStore[pid], plan = [];
    let source = 'fallback';
    if (needsVision) {
      // Text-only models cannot interpret diagrams. Prefer listed vision models, then
      // known vision-capable fallbacks; a stale manual text model is not used for images.
      const baseIds = ms.ids.length ? ms.ids : VISION_PRIOR[pid];
      const visionIds = baseIds.filter((id) => isVisionModel(pid, id));
      const ranked = rankModels(pid, visionIds, ms.stats, ms.unavailable, Date.now());
      if (cfg.modelMode === 'manual' && cfg.manualModel && isVisionModel(pid, cfg.manualModel)) {
        plan.push(cfg.manualModel); source = 'manual vision override';
      } else if (cfg.modelMode === 'manual' && cfg.manualModel) {
        log('warn', pid + ' manual model is not recognized as vision-capable; using a vision model for this image question');
      }
      for (const id of ranked) if (!plan.includes(id)) plan.push(id);
      for (const id of VISION_PRIOR[pid]) if (!plan.includes(id)) plan.push(id);
      if (cfg.fallbackModel && isVisionModel(pid, cfg.fallbackModel) && !plan.includes(cfg.fallbackModel)) plan.push(cfg.fallbackModel);
      if (source === 'fallback') source = plan.length ? 'vision-capable fallback chain' : 'no known vision model';
    } else if (cfg.modelMode === 'manual' && cfg.manualModel) { plan.push(cfg.manualModel); source = 'manual override'; }
    else {
      const ids = ms.ids.length ? ms.ids : PRIOR_MODELS[pid];
      const ranked = rankModels(pid, ids, ms.stats, ms.unavailable, Date.now());
      if (ranked.length) {
        plan.push(ranked[0]);
        const st = ms.stats[ranked[0]];
        source = st && st.n ? 'fastest measured' : ms.ids.length ? 'newest listed, no timing yet' : 'documented default';
        for (const id of ranked.slice(1, 3)) if (!plan.includes(id)) plan.push(id);
        for (const id of PRIOR_MODELS[pid]) if (!plan.includes(id)) plan.push(id);
      }
      if (cfg.fallbackModel && !plan.includes(cfg.fallbackModel)) plan.push(cfg.fallbackModel);
    }
    return { plan, source };
  }

  function gmRequest(o) {
    let handle;
    const promise = new Promise((resolve, reject) => {
      handle = GM_xmlhttpRequest({
        method: o.method || 'POST', url: o.url, headers: o.headers, data: o.body, timeout: o.timeoutMs, responseType: 'text', anonymous: true,
        onload: (r) => resolve({ status: r.status, text: r.responseText || '', headers: r.responseHeaders || '' }),
        onerror: () => reject({ kind: 'network' }),
        ontimeout: () => reject({ kind: 'timeout' }),
        onabort: () => reject({ kind: 'abort' })
      });
    });
    return { promise, abort: () => { try { if (handle && handle.abort) handle.abort(); } catch (e) { /* ignore */ } } };
  }

  function bytesToBase64(buffer) {
    const bytes = new Uint8Array(buffer);
    let binary = '';
    const stride = 0x8000;
    for (let i = 0; i < bytes.length; i += stride) binary += String.fromCharCode.apply(null, bytes.subarray(i, Math.min(i + stride, bytes.length)));
    return btoa(binary);
  }

  function requestImageArrayBuffer(url, ctl) {
    return new Promise((resolve, reject) => {
      let finished = false;
      let handle = null;
      const done = (fn, value) => {
        if (finished) return; finished = true;
        if (ctl && Array.isArray(ctl.imageRequests)) ctl.imageRequests = ctl.imageRequests.filter((x) => x !== handle);
        fn(value);
      };
      try {
        handle = GM_xmlhttpRequest({
          method: 'GET', url, headers: { Accept: 'image/avif,image/webp,image/apng,image/svg+xml,image/*,*/*;q=0.8' },
          responseType: 'arraybuffer', timeout: VISION_DOWNLOAD_TIMEOUT_MS, anonymous: true,
          onload: (r) => {
            if (r.status < 200 || r.status >= 300) return done(reject, new Error('image HTTP ' + r.status));
            const data = r.response;
            if (!data || typeof data.byteLength !== 'number' || !data.byteLength) return done(reject, new Error('empty image response'));
            if (data.byteLength > MAX_VISION_DOWNLOAD_BYTES) return done(reject, new Error('image larger than 8 MB'));
            const head = (r.responseHeaders || '').match(/content-type:\s*([^;\r\n]+)/i);
            const mime = head ? head[1].trim().toLowerCase() : '';
            if (mime && !mime.startsWith('image/') && !/application\/octet-stream/i.test(mime)) return done(reject, new Error('source did not return an image'));
            done(resolve, new Blob([data], { type: mime.startsWith('image/') ? mime : 'application/octet-stream' }));
          },
          onerror: () => done(reject, new Error('could not download image')),
          ontimeout: () => done(reject, new Error('image download timed out')),
          onabort: () => done(reject, new Error('image download cancelled'))
        });
        if (ctl && Array.isArray(ctl.imageRequests)) ctl.imageRequests.push(handle);
      } catch (e) { done(reject, e); }
    });
  }

  async function sourceToBlob(src, ctl) {
    if (/^data:image\//i.test(src) || /^blob:/i.test(src)) {
      const r = await fetch(src);
      if (!r.ok) throw new Error('local image could not be read');
      const blob = await r.blob();
      if (!blob.type.startsWith('image/')) throw new Error('source is not an image');
      if (blob.size > MAX_VISION_DOWNLOAD_BYTES) throw new Error('image larger than 8 MB');
      return blob;
    }
    if (!/^https:\/\//i.test(src) || !isSafeRemoteImageUrl(src)) throw new Error('unsupported or non-public image URL');
    return await requestImageArrayBuffer(src, ctl);
  }

  async function normalizeVisionBlob(blob) {
    let bitmap = null, image = null, objectUrl = '';
    try {
      if (typeof createImageBitmap === 'function') {
        try { bitmap = await createImageBitmap(blob); }
        catch (e) { bitmap = null; }
      }
      if (!bitmap) {
        objectUrl = URL.createObjectURL(blob);
        image = await new Promise((resolve, reject) => {
          const im = new Image(); im.onload = () => resolve(im); im.onerror = () => reject(new Error('image could not be decoded')); im.src = objectUrl;
        });
      }
      const sw = bitmap ? bitmap.width : image.naturalWidth;
      const sh = bitmap ? bitmap.height : image.naturalHeight;
      if (!sw || !sh) throw new Error('image has no readable pixels');
      const scale = Math.min(1, MAX_VISION_DIMENSION / Math.max(sw, sh));
      const canvas = document.createElement('canvas');
      canvas.width = Math.max(1, Math.round(sw * scale)); canvas.height = Math.max(1, Math.round(sh * scale));
      const ctx = canvas.getContext('2d', { alpha: false });
      if (!ctx) throw new Error('image conversion unavailable');
      ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, canvas.width, canvas.height);
      ctx.drawImage(bitmap || image, 0, 0, canvas.width, canvas.height);
      const jpeg = await new Promise((resolve, reject) => canvas.toBlob((b) => b ? resolve(b) : reject(new Error('image compression failed')), 'image/jpeg', 0.86));
      if (!jpeg || !jpeg.size) throw new Error('image compression produced no data');
      return jpeg;
    } finally {
      try { if (bitmap && bitmap.close) bitmap.close(); } catch (e) { /* ignore */ }
      try { if (objectUrl) URL.revokeObjectURL(objectUrl); } catch (e) { /* ignore */ }
    }
  }

  async function hashVisionRefs(refs) {
    const list = refs || [];
    // Prefer SHA-256 via Web Crypto. It is accurate even when a small part of a
    // manual screenshot changes, without a JavaScript loop over every byte.
    try {
      if (window.crypto && window.crypto.subtle && typeof TextEncoder === 'function') {
        const encoder = new TextEncoder(), digests = [];
        for (const ref of list) {
          const src = String(ref.src || '');
          const input = encoder.encode(String(ref.label || '') + '\u241f' + src.length + '\u241f' + src);
          const digest = await window.crypto.subtle.digest('SHA-256', input);
          digests.push(Array.from(new Uint8Array(digest), (b) => b.toString(16).padStart(2, '0')).join(''));
        }
        return digests.join('.');
      }
    } catch (e) { /* fall through to the bounded-work fallback below */ }

    let hash = 2166136261;
    const feed = (str) => { for (let i = 0; i < str.length; i++) { hash ^= str.charCodeAt(i); hash = Math.imul(hash, 16777619); } };
    for (const ref of list) {
      const src = String(ref.src || '');
      feed(String(ref.label || '') + '\u241f' + src.length + '\u241f');
      // A manual image can be a multi-megabyte data URL; sample evenly so the
      // fallback never blocks the UI while still distinguishing typical images.
      const points = Math.min(96, src.length);
      for (let j = 0; j < points; j++) {
        const i = points === 1 ? 0 : Math.floor(j * (src.length - 1) / (points - 1));
        hash ^= src.charCodeAt(i); hash = Math.imul(hash, 16777619);
      }
      hash ^= 255; hash = Math.imul(hash, 16777619);
    }
    return (hash >>> 0).toString(16);
  }

  async function prepareVisionImages(req, refs, ctl) {
    req.images = [];
    req.imageNote = false;
    if (!settings.vision || !refs || !refs.length) return { attached: 0, failed: 0 };
    const unique = [], seen = new Set();
    for (const ref of refs) {
      if (!ref || !validVisionSource(ref.src) || seen.has(ref.src)) continue;
      seen.add(ref.src); unique.push(ref);
      if (unique.length >= MAX_VISION_IMAGES) break;
    }
    let failed = 0;
    const processed = await Promise.all(unique.map(async (ref) => {
      if (ctl && ctl.cancelled) return null;
      try {
        const blob = await sourceToBlob(ref.src, ctl);
        const normalized = await normalizeVisionBlob(blob);
        const data = bytesToBase64(await normalized.arrayBuffer());
        if (!data) throw new Error('image encoding failed');
        return { label: ref.label || 'Question image', mimeType: 'image/jpeg', data };
      } catch (e) {
        failed++;
        log('warn', 'vision image skipped (' + (ref.label || 'image') + '): ' + String(e && e.message || 'read failed').slice(0, 100));
        return null;
      }
    }));
    if (ctl && ctl.cancelled) return { attached: 0, failed };
    req.images = processed.filter(Boolean).slice(0, MAX_VISION_IMAGES);
    req.imageNote = failed > 0 || req.images.length < unique.length;
    if (req.imageNote) log('warn', 'vision: one or more page images could not be attached; the model is instructed not to guess from missing images');
    return { attached: req.images.length, failed };
  }

  }

  // A provider without credentials is skipped instead of producing a guaranteed error.
  // In backend mode the relay holds the keys, so every provider counts as usable.
  function hasCreds(pid) {
    if (settings.backendUrl && pid !== 'nvidia') return true;
    return eligibleKeys(pid).length > 0;
  }

  // Direct mode talks to the provider; backend mode talks to your relay, which holds the keys.
  function buildHttp(pid, kind, model, req, creds, opts) {
    const ad = adapters[pid];
    if (creds.mode === 'backend') {
      const headers = { 'Content-Type': 'application/json' };
      if (creds.secret) headers.Authorization = 'Bearer ' + creds.secret;
      if (kind === 'list') return { method: 'GET', url: settings.backendUrl + '/v1/models/' + pid, headers };
      return { method: 'POST', url: settings.backendUrl + '/v1/generate/' + pid, headers, body: JSON.stringify({ model, body: ad.body(req, model, settings, opts) }) };
    }
    if (kind === 'list') return { method: 'GET', url: typeof ad.listUrl === 'function' ? ad.listUrl() : ad.listUrl, headers: ad.auth(creds.secret) };
    return { method: 'POST', url: ad.genUrl(model), headers: Object.assign({ 'Content-Type': 'application/json' }, ad.auth(creds.secret)), body: JSON.stringify(ad.body(req, model, settings, opts)) };
  }

  function httpError(status, text, headers) {
    let detail = '', code = '';
    try {
      const j = JSON.parse(text);
      detail = (j.error && j.error.message) || j.message || (typeof j.detail === 'string' ? j.detail : '');
      code = (j.error && j.error.status) || j.status || '';
    } catch (e) { /* not JSON */ }
    detail = redact(String(detail), secrets()).slice(0, 200);
    code = String(code);
    const tail = detail ? ' Server said: ' + detail : '';
    const ra = /retry-after:\s*(\d+)/i.exec(headers || '');
    if (status === 401) return fail('auth', 'Authentication failed (HTTP 401). Check the key or token.' + tail, false);
    if (status === 403) return fail('auth', 'The provider refused this request (HTTP 403). The key may lack access to this model, or the model may be blocked in your region.' + tail, false);
    if (/API_KEY_INVALID|API key not valid|invalid api key/i.test(detail) || /API_KEY_INVALID/.test(code)) {
      return fail('auth', 'The provider rejected this API key. Save a valid key in Settings.' + tail, false);
    }
    if (status === 404 || /model.*(not found|decommission|does not exist|not supported)/i.test(detail)) return fail('not-found', 'Model not available (HTTP ' + status + ').' + tail, false);
    if (status === 429) {
      const e = fail('rate-limit', 'Rate limit or quota reached (HTTP 429). Free tiers have per-minute limits; wait or use another key.' + tail, true);
      e.retryAfterMs = ra ? Number(ra[1]) * 1000 : 0;
      return e;
    }
    if (status === 402) {
      const e = fail('quota', 'The provider says this account has no credits left (HTTP 402). Add credits or use another key.' + tail, true);
      e.retryAfterMs = ra ? Number(ra[1]) * 1000 : 0;
      return e;
    }
    if (status === 400) return fail('bad-request', 'The service rejected the request (HTTP 400). An invalid API key can also return 400.' + tail, false);
    if (status >= 500) return fail('server', 'The service had an error (HTTP ' + status + ').' + tail, true);
    return fail('http', 'Unexpected HTTP status ' + status + '.' + tail, false);
  }

  // One model, bounded retries. Resolves to a validated result or throws {code,message,retryable}.
  async function callModel(pid, model, req, c, creds, retries) {
    const ad = adapters[pid], max = retries === undefined ? settings.maxRetries : retries;
    const opts = { thinking: true, noJson: false };
    let last;
    for (let attempt = 0; attempt <= max; attempt++) {
      if (c.cancelled) throw fail('cancelled', 'Cancelled.', false);
      S.analysis.attempt = Math.max(S.analysis.attempt, attempt + 1); ui.refresh('analysis');
      try {
        const http = buildHttp(pid, 'gen', model, req, creds, opts);
        const r = gmRequest({ method: 'POST', url: http.url, headers: http.headers, body: http.body, timeoutMs: settings.timeoutSec * 1000 });
        c.abort = r.abort;
        let res;
        try { res = await r.promise; } catch (e) {
          if (e && e.kind === 'abort') throw fail('cancelled', 'Cancelled.', false);
          if (e && e.kind === 'timeout') throw fail('timeout', 'No response within ' + settings.timeoutSec + ' seconds.', true);
          throw fail('network', 'Network error. Check your connection and accept Tampermonkey\'s @connect prompt if shown.', true);
        }
        if (c.cancelled) throw fail('cancelled', 'Cancelled.', false);
        if (res.status !== 200) {
          const he = httpError(res.status, res.text, res.headers);
          if (pid === 'gemini' && opts.thinking && he.code === 'bad-request' && /thinking/i.test(he.message)) { opts.thinking = false; he.retryNow = true; }
          if ((pid === 'openrouter' || pid === 'groq') && !opts.noJson && he.code === 'bad-request' && /response_format|json mode|json_object/i.test(he.message)) {
            opts.noJson = true; he.retryNow = true;
          }
          throw he;
        }
        let json; try { json = JSON.parse(res.text); } catch (e) { throw fail('bad-json', 'The service returned a non-JSON response.', true); }
        const raw = ad.extract(json);
        const obj = typeof raw === 'string' ? parseJsonLoose(raw) : raw;
        if (obj == null) throw fail('bad-json', 'The model did not return valid JSON.', true);
        const v = validateAnalysis(obj, req);
        if (!v.ok) throw fail('invalid-response', 'Response rejected: ' + v.error + '.', true);
        if (creds.keyId) noteKeyOk(pid, creds.keyId);
        return v.value;
      } catch (e) {
        last = e && e.code ? e : fail('unknown', 'Unexpected error: ' + (e && e.message ? e.message : 'unknown'), false);
        log('warn', pid + ' ' + model + ' attempt ' + (attempt + 1) + ': ' + last.code);
        if (last.retryNow) { attempt--; continue; }
        // Rate limits and empty quotas are handled by the key failover loop, not here:
        // waiting on a throttled key would delay a healthy one.
        if (last.code === 'rate-limit' || last.code === 'quota') throw last;
        if (!last.retryable || attempt === max || c.cancelled) throw last;
        const backoff = Math.min(30000, Math.max(1000 * Math.pow(2, attempt), last.retryAfterMs || 0));
        await new Promise((res) => later(res, backoff + Math.round(Math.random() * 350)));
      }
    }
    throw last;
  }

  const newRow = (pid) => ({ provider: pid, model: '', startAt: null, latencyMs: null, validation: 'pending', failure: '', role: 'pending' });

  // Never rejects. Returns {provider, model, status, value|failure+message, latencyMs}.
  // Keys are tried in pool order. A throttle marks that key with a cooldown and
  // moves on; an auth failure marks it rejected; a healthy key clears both.
  async function runProvider(pid, req, c, rows, onlyKeyId) {
    const row = rows[pid] || (rows[pid] = newRow(pid));
    const t0 = performance.now();
    row.startAt = Date.now(); row.latencyMs = null; row.validation = 'pending'; row.failure = '';
    const finish = (rec) => { row.latencyMs = Math.round(performance.now() - t0); return Object.assign(rec, { provider: pid, latencyMs: row.latencyMs, model: row.model }); };
    let keys;
    if (settings.backendUrl && pid !== 'nvidia') keys = [null];
    else if (onlyKeyId) keys = poolFor(pid).filter((k) => k.id === onlyKeyId);
    else keys = eligibleKeys(pid);
    if (!keys.length) {
      row.validation = 'not sent'; row.failure = 'no-key';
      const pool = poolFor(pid);
      const msg = pool.length
        ? 'No usable ' + adapters[pid].label + ' key: every saved key is disabled, cooling down, or rejected. Open Settings, test a key, or retry after the cooldown.'
        : 'No ' + adapters[pid].label + ' key saved. Add one in Settings.';
      return finish({ status: 'error', failure: 'no-key', message: msg });
    }
    const needsVision = !!(settings.vision && req.images && req.images.length);
    const plan = modelPlan(pid, needsVision).plan, ms = modelStore[pid];
    let last = null, nextKey = false;
    for (const key of keys) {
      if (c.cancelled) break;
      let creds;
      try { creds = credsFor(pid, key); } catch (e) { last = e; break; }
      nextKey = false;
      for (const model of plan) {
        if (c.cancelled) break;
        row.model = model;
        try {
          const value = await callModel(pid, model, req, c, creds);
          updateStat(ms.stats, model, performance.now() - t0, true); saveModels();
          row.validation = value.uncertain ? 'valid, uncertain' : 'valid, ' + value.confidence + ' confidence';
          return finish({ status: 'ok', value });
        } catch (e) {
          last = e;
          if (e.code === 'cancelled') break;
          if (e.code === 'not-found') { ms.unavailable[model] = Date.now(); saveModels(); log('warn', pid + ' model unavailable: ' + model); continue; }
          if (e.code === 'invalid-response' || e.code === 'bad-json') { updateStat(ms.stats, model, 0, false); saveModels(); }
          if (e.code === 'auth' && key) {
            noteKeyFailure(pid, key.id, 'auth', 0, true);
            log('warn', pid + ' key ' + maskKey(key.key) + ' rejected; trying next key if any');
            nextKey = true; break;
          }
          if ((e.code === 'rate-limit' || e.code === 'quota') && key) {
            const wait = Math.min(3600000, e.retryAfterMs || (e.code === 'quota' ? 900000 : 60000));
            noteKeyFailure(pid, key.id, e.code, wait);
            log('warn', pid + ' key ' + maskKey(key.key) + ' cooling down ' + Math.round(wait / 1000) + 's (' + e.code + ')');
            nextKey = true; break;
          }
          // Overloaded (503) or throttled (429) models are common and transient; another model
          // from the same provider usually answers. Auth and input problems affect every model.
          if (e.code === 'server' || e.code === 'rate-limit' || e.code === 'quota') { log('warn', pid + ' ' + model + ' unavailable (' + e.code + '), trying next model'); continue; }
          break;
        }
      }
      if (c.cancelled) break;
      if (!nextKey) break;
    }
    last = last || fail('cancelled', 'Cancelled.', false);
    row.validation = last.code === 'invalid-response' ? 'invalid' : last.code === 'cancelled' ? 'cancelled' : 'no valid response';
    row.failure = last.code;
    return finish({ status: 'error', failure: last.code, message: last.message });
  }

  const inflightRefresh = {};
  async function refreshModels(pid, force) {
    const ms = modelStore[pid];
    if (!force && ms.fetchedAt && Date.now() - ms.fetchedAt < settings.modelRefreshHours * 3600000) return { ok: true, cached: true };
    if (inflightRefresh[pid]) return inflightRefresh[pid];
    inflightRefresh[pid] = (async () => {
      try {
        const creds = credsFor(pid);
        const http = buildHttp(pid, 'list', '', null, creds, {});
        const res = await gmRequest({ method: 'GET', url: http.url, headers: http.headers, timeoutMs: 15000 }).promise;
        if (res.status !== 200) return { ok: false, message: httpError(res.status, res.text, res.headers).message };
        let json; try { json = JSON.parse(res.text); } catch (e) { return { ok: false, message: 'Model list was not valid JSON.' }; }
        const ids = modelCandidates(pid, json);
        if (!ids.length) return { ok: false, message: 'The provider returned no suitable chat models.' };
        ms.ids = ids; ms.fetchedAt = ms.verifiedAt = Date.now(); ms.unavailable = {};
        saveModels(); ui.refresh('models');
        return { ok: true, count: ids.length };
      } catch (e) {
        return { ok: false, message: e && e.message ? e.message : e && e.kind === 'timeout' ? 'Model list request timed out.' : 'Could not fetch the model list.' };
      } finally { delete inflightRefresh[pid]; }
    })();
    return inflightRefresh[pid];
  }

  const trivial = () => buildRequest('Which of these two numbers is larger?', ['2', '9'], '').req;
  async function testProvider(pid, keyId) {
    S.test[pid] = { status: 'busy', msg: 'Testing\u2026' }; ui.refresh('test');
    const rows = {};
    const rec = await runProvider(pid, trivial(), { cancelled: false, abort() {} }, rows, keyId);
    S.test[pid] = rec.status === 'ok'
      ? { status: 'ok', msg: 'Connected: ' + rec.model + ' answered in ' + rec.latencyMs + ' ms (response validated).' }
      : { status: 'fail', msg: rec.message };
    ui.refresh('test');
  }

  async function benchmarkModels(pid, keyId) {
    S.test[pid] = { status: 'busy', msg: 'Benchmarking up to 3 models\u2026' }; ui.refresh('test');
    const key = keyId ? keyById(pid, keyId) : eligibleKeys(pid)[0];
    let creds;
    try { creds = credsFor(pid, key); } catch (e) { S.test[pid] = { status: 'fail', msg: e.message }; ui.refresh('test'); return; }
    await refreshModels(pid, true);
    const ms = modelStore[pid];
    const ids = rankModels(pid, ms.ids.length ? ms.ids : PRIOR_MODELS[pid], {}, ms.unavailable, Date.now()).slice(0, 3);
    const out = [];
    for (const id of ids) {
      const t0 = performance.now();
      try {
        await callModel(pid, id, trivial(), { cancelled: false, abort() {} }, creds, 0);
        const ms2 = Math.round(performance.now() - t0); updateStat(ms.stats, id, ms2, true); out.push({ id, ok: true, ms: ms2 });
      } catch (e) {
        if (e.code === 'not-found') ms.unavailable[id] = Date.now(); else updateStat(ms.stats, id, 0, false);
        out.push({ id, ok: false, code: e.code });
      }
    }
    saveModels();
    const best = out.filter((x) => x.ok).sort((a, b) => a.ms - b.ms)[0];
    S.test[pid] = {
      status: best ? 'ok' : 'fail',
      msg: out.map((x) => x.id + ': ' + (x.ok ? x.ms + ' ms' : 'failed (' + x.code + ')')).join('; ') + (best ? '. Fastest valid: ' + best.id + '.' : '')
    };
    ui.refresh('test'); ui.refresh('models');
  }

  /* ====================================================================
   * Race orchestration, cache, history
   * ==================================================================== */
  const cache = new Map();
  const cacheGet = (k) => {
    const e = cache.get(k);
    if (!e) return null;
    if (Date.now() - e.t > settings.cacheTtlMin * 60000) { cache.delete(k); return null; }
    return e.v;
  };
  const cacheSet = (k, v) => { cache.set(k, { t: Date.now(), v }); if (cache.size > 100) cache.delete(cache.keys().next().value); };

  function cancelAnalysis() {
    clearAutoTimer();
    if (current) { current.cancelled = true; try { current.abort(); } catch (e) { /* ignore */ } current = null; }
    if (S.analysis.status === 'loading') S.analysis.status = 'idle';
  }

  let runSeq = 0;
  const blankAnalysis = () => ({ status: 'idle', key: '', sig: '', result: null, error: null, info: null, fromCache: false, highlight: '', attempt: 0 });

  async function runAnalysis(force) {
    if (!settings.acknowledged) { S.analysis = Object.assign(blankAnalysis(), { status: 'error', error: fail('ack', 'Confirm the use notice on the Overview tab first.', false) }); ui.refresh('analysis'); return; }
    const b = buildRequest(S.edit.question, S.edit.choices.split('\n'), S.edit.notes);
    if (!b.ok) { clearHighlights(); S.analysis = Object.assign(blankAnalysis(), { status: 'error', error: fail('input', b.error, false) }); ui.refresh('analysis'); return; }
    const liveDetection = detectNow();
    const detectionMatchesRequest = !!(liveDetection && liveDetection.ok && liveDetection.sig === b.sig);
    const detectedImageRefs = detectionMatchesRequest ? (liveDetection.imageRefs || []) : [];
    const pageImageRefs = settings.vision ? detectedImageRefs : [];
    const manualImageRefs = settings.vision && Array.isArray(S.edit.images) ? S.edit.images : [];
    // User-attached images take priority over auto-detected candidates, especially if a CDN blocks access.
    const allImageRefs = [...manualImageRefs, ...pageImageRefs];
    const imageRefs = allImageRefs.slice(0, MAX_VISION_IMAGES);
    const omittedImageRefs = allImageRefs.length > MAX_VISION_IMAGES;
    const hasImagePlaceholder = /\[(?:Question image|Image answer)/i.test(b.req.question + ' ' + b.req.choices.map((x) => x.text).join(' '));
    const expectsImage = hasImagePlaceholder || detectedImageRefs.length > 0;
    if ((expectsImage && (!settings.vision || !imageRefs.length)) || (settings.vision && omittedImageRefs)) b.req.imageNote = true;
    const imageSig = settings.vision ? await hashVisionRefs(imageRefs) : 'vision-off';
    const allEnabled = PROVIDER_IDS.filter((p) => settings.providers[p].enabled);
    const preferNvidia = settings.preferNvidiaVision && settings.vision && imageRefs.length > 0 && !settings.verify.enabled &&
      settings.providers.nvidia.enabled && hasCreds('nvidia');
    // Route picture-bearing questions to Nemotron when preferred and configured.
    // Text-only questions and explicit verification continue using the regular provider pool.
    const enabled = preferNvidia ? ['nvidia'] : allEnabled;
    if (!enabled.length) { S.analysis = Object.assign(blankAnalysis(), { status: 'error', error: fail('no-provider', 'All providers are switched off. Enable one in Settings.', false) }); ui.refresh('analysis'); return; }
    const pool = settings.verify.enabled ? enabled.filter((p) => settings.verify.participants.includes(p)) : enabled;
    if (settings.verify.enabled && pool.length < 2) {
      S.analysis = Object.assign(blankAnalysis(), { status: 'error', error: fail('verify-config', 'Verification needs at least two enabled participant providers. Enable more providers or turn verification off.', false) }); ui.refresh('analysis'); return;
    }
    const pids = pool.filter(hasCreds);
    if (!pids.length) {
      const msg = settings.backendUrl
        ? 'No backend token saved. Add one in Settings.'
        : 'No usable API key for ' + pool.map((p) => adapters[p].label).join(', ') + '. Add a key or wait for a cooldown in Settings.';
      S.analysis = Object.assign(blankAnalysis(), { status: 'error', error: fail('no-key', msg, false) }); ui.refresh('analysis'); return;
    }
    if (settings.verify.enabled && pids.length < settings.verify.threshold) {
      log('warn', 'verification: only ' + pids.length + ' participant(s) have a usable key; ' + settings.verify.threshold + ' agreeing answers are required');
    }
    const policy = settings.verify.enabled
      ? { mode: 'verify', threshold: settings.verify.threshold, minConfidence: settings.policy.minConfidence }
      : settings.policy;
    const key = pids.join('+') + '|' + policy.mode + '|' + (policy.threshold || '') + '|' + policy.minConfidence + '|' + b.sig + '|' + b.req.notes + '|img:' + imageSig;
    if (S.analysis.status === 'loading' && S.analysis.key === key) return; // duplicate request
    if (!force && settings.cache) {
      const hit = cacheGet(key);
      if (hit) { S.analysis = Object.assign(blankAnalysis(), { status: 'done', key, sig: b.sig, result: hit.result, info: hit.info, fromCache: true }); applyHighlight(); ui.refresh('analysis'); maybeAutoSelect(); return; }
    }
    cancelAnalysis(); clearHighlights();
    const ctl = { cancelled: false, imageRequests: [], abort() { for (const r of this.imageRequests.splice(0)) { try { if (r && r.abort) r.abort(); } catch (e) { /* ignore */ } } } };
    current = ctl;
    const run = { id: ++runSeq, startedAt: Date.now(), sig: b.sig, rows: {}, outcome: 'preparing', firstUsableMs: null, pids, preparingImages: !!(settings.vision && imageRefs.length), imageCount: 0 };
    pids.forEach((p) => { run.rows[p] = newRow(p); });
    S.run = run;
    S.analysis = Object.assign(blankAnalysis(), { status: 'loading', key, sig: b.sig, attempt: 1 });
    ui.refresh('analysis');

    if (settings.vision && imageRefs.length) {
      const imageStatus = await prepareVisionImages(b.req, imageRefs, ctl);
      if (current !== ctl || ctl.cancelled) { run.outcome = 'cancelled'; ui.refresh('analysis'); return; }
      run.imageCount = imageStatus.attached;
      run.preparingImages = false;
      if ((!imageStatus.attached && imageRefs.length) || omittedImageRefs) b.req.imageNote = true;
      ui.refresh('analysis');
    }
    run.outcome = 'running';
    const out = await raceProviders(pids, (pid, c) => runProvider(pid, b.req, c, run.rows), policy, ctl);
    if (current !== ctl || ctl.cancelled || out.kind === 'cancelled') { run.outcome = 'cancelled'; ui.refresh('analysis'); return; }
    current = null;
    run.outcome = out.kind;
    for (const p of pids) run.rows[p].role = out.roles[p] || 'n/a';
    const usable = (out.finished || []).filter((r) => isAcceptable(r, policy)).map((r) => r.latencyMs);
    run.firstUsableMs = usable.length ? Math.min.apply(null, usable) : null;
    const base = { key, sig: b.sig, attempt: S.analysis.attempt };

    if (out.kind === 'accepted') {
      const w = out.winner;
      const info = {
        kind: 'accepted', provider: w.provider, model: w.model, latencyMs: w.latencyMs,
        agreed: !!out.agreed, single: !!out.single, verified: !!out.verified,
        agreement: out.agreement ? { letter: out.agreement.letter, count: out.agreement.count, of: out.agreement.of } : null
      };
      S.analysis = Object.assign(blankAnalysis(), base, { status: 'done', result: w.value, info });
      if (settings.cache) cacheSet(key, { result: w.value, info });
      addHistory(b.req, w.value, b.sig, info);
    } else if (out.kind === 'unverified') {
      const w = out.best;
      const info = {
        kind: 'unverified', provider: w.provider, model: w.model, latencyMs: w.latencyMs, verified: false,
        agreement: out.agreement ? { letter: out.agreement.letter, count: out.agreement.count, of: out.agreement.of } : null,
        note: 'Usable answer, but only ' + (out.agreement ? out.agreement.count : 1) + ' of ' + (out.agreement ? out.agreement.of : 1) + ' models agreed; ' + (policy.threshold || 2) + ' required.'
      };
      S.analysis = Object.assign(blankAnalysis(), base, { status: 'done', result: Object.assign({}, w.value, { uncertain: w.value.uncertain }), info });
      addHistory(b.req, w.value, b.sig, info);
    } else if (out.kind === 'inconclusive') {
      const w = out.best;
      const info = { kind: 'inconclusive', provider: w.provider, model: w.model, latencyMs: w.latencyMs, note: 'No response met your minimum confidence (' + policy.minConfidence + ').' };
      S.analysis = Object.assign(blankAnalysis(), base, { status: 'done', result: Object.assign({}, w.value, { uncertain: true }), info });
    } else if (out.kind === 'conflict') {
      const info = {
        kind: 'conflict',
        conflict: out.conflict.map((r) => ({ provider: r.provider, model: r.model, choice: r.value.suggestedChoice, text: r.value.suggestedChoiceText, confidence: r.value.confidence, latencyMs: r.latencyMs })),
        agreement: out.agreement ? { letter: out.agreement.letter, count: out.agreement.count, of: out.agreement.of } : null
      };
      S.analysis = Object.assign(blankAnalysis(), base, { status: 'done', result: null, info });
    } else {
      const msg = (out.finished || []).map((r) => adapters[r.provider].label + ': ' + r.message).join(' | ') || 'No provider responded.';
      S.analysis = Object.assign(blankAnalysis(), base, { status: 'error', error: fail('all-failed', msg, true) });
    }
    applyHighlight();
    ui.refresh('analysis');
    maybeAutoSelect();
  }

  function addHistory(req, result, sig, info) {
    if (!settings.history) return;
    S.history = S.history.filter((e) => e.sig !== sig);
    S.history.unshift({
      id: Date.now().toString(36) + Math.random().toString(36).slice(2, 6), ts: Date.now(), sig,
      question: req.question, choices: req.choices.map((c) => c.text), result, model: info.provider + ':' + info.model
    });
    S.history = S.history.slice(0, 200);
    store.set(K.history, JSON.stringify(S.history));
    ui.refresh('history');
  }

    function viewAnalysis() {
      const q = h('textarea', { rows: 3, value: S.edit.question, placeholder: 'Question text (paste one if detection fails)' });

      const updateImageNote = () => {
        const count = (S.edit.images || []).length;
        imageNote.textContent = count ? count + ' manual image attachment' + (count === 1 ? '' : 's') + ' ready. Page images are also detected automatically.' : 'No manual attachments. Page question/answer images are detected automatically when possible.';
      };
      imagePicker.addEventListener('change', async () => {
        const files = Array.from(imagePicker.files || []);
        let added = 0;
        for (const file of files) {
          if (!file.type.startsWith('image/')) continue;
          if (file.size > 8 * 1024 * 1024) { imageNote.textContent = file.name + ' is larger than 8 MB and was skipped.'; continue; }
          if ((S.edit.images || []).length >= MAX_VISION_IMAGES) { imageNote.textContent = 'Maximum of ' + MAX_VISION_IMAGES + ' manual images reached.'; break; }
          try {
            const data = await new Promise((resolve, reject) => {
              const reader = new FileReader(); reader.onload = () => resolve(String(reader.result || '')); reader.onerror = () => reject(new Error('read failed')); reader.readAsDataURL(file);
            });
            if (!/^data:image\//i.test(data)) continue;
            S.edit.images.push({ label: 'Manual image ' + (S.edit.images.length + 1), src: data }); added++;
          } catch (e) { imageNote.textContent = 'Could not read ' + file.name + '.'; }
        }
        imagePicker.value = '';
        updateImageNote();
        if (added) imageNote.textContent = added + ' image' + (added === 1 ? '' : 's') + ' attached. They will be resized before analysis.';
      });
      q.addEventListener('input', () => { S.edit.question = q.value; S.edit.dirty = true; });
      c.addEventListener('input', () => { S.edit.choices = c.value; S.edit.dirty = true; });
      n.addEventListener('input', () => { S.edit.notes = n.value; });
      refs = { q, c, n, imagePicker, imageNote, res: h('div', { class: 'stack', 'aria-live': 'polite' }) };
      updateImageNote();
      const lab = (t, el) => h('div', {}, h('div', { class: 'k', text: t }), el);

    function syncTabs() {
      if (!tabsEl) return;
      const max = tabsEl.scrollWidth - tabsEl.clientWidth;
      tabsEl.classList.toggle('can-l', tabsEl.scrollLeft > 2);
      tabsEl.classList.toggle('can-r', tabsEl.scrollLeft < max - 2);
      const sel = tabsEl.querySelector('[aria-selected=true]');
      if (!sel) return;
      const l = sel.offsetLeft, r = l + sel.offsetWidth;
      if (l < tabsEl.scrollLeft) tabsEl.scrollLeft = Math.max(0, l - 8);
      else if (r > tabsEl.scrollLeft + tabsEl.clientWidth) tabsEl.scrollLeft = r - tabsEl.clientWidth + 8;
    }

    function flush() {
      if (raf) { try { cancelAnimationFrame(raf); } catch (e) { /* ignore */ } raf = 0; }
      if (rafTimer) { clearTimeout(rafTimer); rafTimer = 0; }
      if (!win) return;
      dot.dataset.s = S.analysis.status;
      const t = S.tab;
      if (t === 'overview') renderTab();
      else if (t === 'analysis') { if (kinds.has('detect')) syncInputs(); if (kinds.has('detect') || kinds.has('analysis')) renderResult(); }
      else if (t === 'diag' && (kinds.has('analysis') || kinds.has('detect'))) renderTab();
      else if (t === 'history' && kinds.has('history')) renderHistoryList();
      if (t === 'ai' && (kinds.has('test') || kinds.has('models'))) renderTab();
      kinds.clear();
    }
    // requestAnimationFrame alone is not reliable: it is paused in background tabs and can be
    // dropped while a page is still painting, which used to leave the panel frozen. A timer
    // backstop guarantees the frame runs at most ~200 ms late.
    function refresh(kind) {
      kinds.add(kind);
      if (!raf) raf = requestAnimationFrame(flush);
      if (!rafTimer) rafTimer = setTimeout(flush, 200);
    }

        el.setPointerCapture(e.pointerId);
        const sx = e.clientX, sy = e.clientY, p = settings.panel, o = { x: p.x, y: p.y, w: p.w, h: p.h };
        const r = win.getBoundingClientRect();
        if (p.x < 0) o.x = r.left;
        const move = (ev) => onMove(ev.clientX - sx, ev.clientY - sy, o);
        const up = () => { el.removeEventListener('pointermove', move); el.removeEventListener('pointerup', up); el.removeEventListener('pointercancel', up); onEnd(); };
        el.addEventListener('pointermove', move); el.addEventListener('pointerup', up); el.addEventListener('pointercancel', up);
      });
    }
    const ibtn = (label, glyph, fn) => h('button', { class: 'ib', type: 'button', 'aria-label': label, title: label, onclick: fn, text: glyph });

    function mount() {
      host = h('div', { id: HOST_ID });
      shadow = host.attachShadow({ mode: 'open' });
      dot = h('span', { class: 'dot', 'data-s': 'idle', 'aria-hidden': 'true' });
      bar = h('div', { class: 'bar', tabindex: '0', role: 'toolbar', 'aria-label': 'Window controls. Alt plus arrow keys move the window.' },
        h('span', { class: 'brand', 'aria-hidden': 'true', text: 'AI' }),
        h('div', { class: 'titles' },
          h('span', { class: 'title', text: 'Aureus External' }),
          h('span', { class: 'tag', text: 'Wayground + Kahoot' })),
        h('span', { class: 'sp' }), dot,
        ibtn('Collapse or expand', '\u25BE', () => { settings.panel.collapsed = !settings.panel.collapsed; changed(); }),
        ibtn('Minimize', '\u2013', () => { settings.panel.minimized = true; changed(); }),
        ibtn('Maximize or restore', '\u25A2', () => { settings.panel.maximized = !settings.panel.maximized; changed(); }),
        ibtn('Close dashboard', '\u2715', () => { settings.enabled = false; saveSettings(); applyEnabled(); }));
      tabsEl = h('div', { class: 'tabs', role: 'tablist', 'aria-label': 'Dashboard sections' }, TABS.map(([id, t]) =>
        h('button', { role: 'tab', id: 'tab-' + id, 'data-id': id, type: 'button', text: t, onclick: () => setTab(id) })));
      tabsEl.addEventListener('scroll', syncTabs, { passive: true });
      tabsEl.addEventListener('keydown', (e) => {
        if (e.key !== 'ArrowRight' && e.key !== 'ArrowLeft') return;
        const i = TABS.findIndex((t) => t[0] === S.tab), n = (i + (e.key === 'ArrowRight' ? 1 : TABS.length - 1)) % TABS.length;
        setTab(TABS[n][0]); tabsEl.querySelector('[data-id=' + TABS[n][0] + ']').focus(); e.preventDefault();
      });
      bodyEl = h('div', { class: 'body', role: 'tabpanel', tabindex: '0' });
      grip = h('div', { class: 'grip', role: 'separator', 'aria-label': 'Resize' });
      win = h('div', { class: 'win', role: 'dialog', 'aria-label': 'Aureus External' }, h('div', { class: 'glow' }), h('div', { class: 'glass' }), h('div', { class: 'bgwrap' }, h('div', { class: 'bgimg' }), h('div', { class: 'bgdim' })), h('div', { class: 'ring' }), bar, tabsEl, bodyEl, grip);
      pill = h('button', { class: 'pill', type: 'button', 'aria-label': 'Restore Aureus External', text: 'Au', onclick: () => { settings.panel.minimized = false; changed(); } });
      let sheeted = false;
      try { const sh = new CSSStyleSheet(); sh.replaceSync(CSS); shadow.adoptedStyleSheets = [sh]; sheeted = true; } catch (e) { /* fall back */ }
      if (!sheeted) shadow.append(h('style', { text: CSS }));
      shadow.append(win, pill);
      document.documentElement.append(host);

      drag(bar, (dx, dy, o) => { settings.panel.x = o.x + dx; settings.panel.y = o.y + dy; layout(); },
        () => { const r = win.getBoundingClientRect(); settings.panel.x = Math.round(r.left); settings.panel.y = Math.round(r.top); saveSettings(); });
      drag(grip, (dx, dy, o) => { settings.panel.w = o.w + dx; settings.panel.h = o.h + dy; layout(); },
        () => { settings = validateSettings(settings); saveSettings(); });
      bar.addEventListener('keydown', (e) => {
        if (!e.altKey) return;
        const m = { ArrowLeft: [-12, 0], ArrowRight: [12, 0], ArrowUp: [0, -12], ArrowDown: [0, 12] }[e.key];
        if (!m) return;
        const r = win.getBoundingClientRect(); settings.panel.x = r.left + m[0]; settings.panel.y = r.top + m[1]; layout(); saveSettings(); e.preventDefault();
      });
      listen(window, 'resize', debounce(layout, 100));
      applyAppearance(); layout(); renderTab();
    }
    function open() {
      if (!host) return;
      host.style.display = ''; layout(); applyAppearance();
      if (win) win.classList.remove('no-fade');
      const show = () => { if (win) win.classList.add('is-open'); };
      requestAnimationFrame(show);
      later(show, 90); // in case the animation frame is throttled
      // If the opening transition never advances (throttled tab, animations disabled by an
      // extension), drop it so the panel cannot stay stuck at opacity 0 and unclickable.
      later(() => {
        if (!win || !win.classList.contains('is-open')) return;
        if (parseFloat(getComputedStyle(win).opacity) < 0.9) win.classList.add('no-fade');
      }, 400);
    }
    function close() {
      if (!win) return;
      win.classList.remove('is-open', 'no-fade');
      pill.style.display = 'none';
      later(() => { if (!settings.enabled && host) host.style.display = 'none'; }, win.dataset.anim === 'off' ? 0 : 220);
    }
    function unmount() { if (raf) cancelAnimationFrame(raf); if (host) host.remove(); host = win = null; }

    return { mount, open, close, unmount, refresh, applyAppearance, layout };
  })();

  /* ====================================================================
   * Lifecycle
   * ==================================================================== */
  function applyEnabled() {
    if (settings.enabled) { ui.open(); startDetection(); }
    else { ui.close(); stopDetection(); cancelAnalysis(); clearHighlights(); }
    saveSettings();
  }

  function destroy() {
    stopDetection(); cancelAnalysis(); clearHighlights();
    saveSettings.cancel(); flushSettings();
    cleanups.splice(0).forEach((f) => { try { f(); } catch (e) { /* ignore */ } });
    timers.forEach((t) => clearTimeout(t)); timers.clear();
    ui.unmount();
    removePageCss();
    delete window.__wsa;
  }

  function init() {
    for (const pid of PROVIDER_IDS) if (seedPresetKey(pid)) log('info', 'Bundled ' + pid + ' key applied');
    updatePageStyle();
    ui.mount();
    if (settings.enabled) { ui.open(); startDetection(); } else { ui.close(); }
    listen(window, 'popstate', scheduleScan);
    listen(window, 'hashchange', scheduleScan);
    listen(window, 'load', scheduleScan);
    listen(document, 'visibilitychange', () => { if (!document.hidden) scheduleScan(); });
    try { if (document.fonts && document.fonts.ready) document.fonts.ready.then(() => scheduleScan()); } catch (e) { /* ignore */ }
    listen(window, 'keydown', (e) => {
      if (e.altKey && e.shiftKey && e.code === 'KeyW') { e.preventDefault(); settings.enabled = !settings.enabled; applyEnabled(); ui.refresh('detect'); }
      if (e.altKey && e.shiftKey && e.code === 'KeyG') { e.preventDefault(); runAnalysis(false); }
    });
    listen(window, 'pagehide', () => { flushSettings(); stopDetection(); cancelAnalysis(); });
    listen(window, 'pageshow', (e) => { if (e.persisted) startDetection(); });
    try {
      GM_registerMenuCommand('Open Aureus External', () => { settings.enabled = true; settings.panel.minimized = false; applyEnabled(); ui.layout(); });
      GM_registerMenuCommand('Reset dashboard position', () => { settings.panel.x = -1; settings.panel.y = 72; settings.panel.maximized = false; settings.panel.minimized = false; saveSettings(); ui.layout(); });
    } catch (e) { /* menu unavailable */ }
    if (settings.enabled) later(() => PROVIDER_IDS.forEach((pid) => { if (settings.providers[pid].enabled) refreshModels(pid, false); }), 1500);
    window.__wsa = { destroy, version: VERSION };
    log('info', 'Initialized v' + VERSION);
  }

  init();
})();
